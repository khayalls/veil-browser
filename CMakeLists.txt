cmake_minimum_required(VERSION 3.25)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Local Mac architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Deployment target")
set(CMAKE_Swift_FLAGS "-module-cache-path ${CMAKE_BINARY_DIR}/ModuleCache" CACHE STRING "Swift flags")
project(VEIL VERSION 0.1.0 LANGUAGES C CXX OBJCXX Swift)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_Swift_LANGUAGE_VERSION 5)
set(PROJECT_ARCH arm64)
if(NOT CEF_ROOT)
  get_filename_component(CEF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.work/dependencies/cef" ABSOLUTE)
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
file(GLOB SWIFT_SOURCES CONFIGURE_DEPENDS "Sources/*.swift")
file(GLOB_RECURSE VEIL_FINGERPRINT_INPUTS CONFIGURE_DEPENDS "Sources/*" "Resources/*" "scripts/*.sh" "CMakeLists.txt")
list(SORT VEIL_FINGERPRINT_INPUTS)
set(VEIL_FINGERPRINT "${CMAKE_GENERATOR};${CMAKE_BUILD_TYPE};${PROJECT_VERSION}")
foreach(input ${VEIL_FINGERPRINT_INPUTS})
  file(SHA256 "${input}" input_hash)
  string(APPEND VEIL_FINGERPRINT "${input_hash}")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${VEIL_FINGERPRINT_INPUTS})
string(SHA256 VEIL_BUILD_HASH "${VEIL_FINGERPRINT}")
string(SUBSTRING "${VEIL_BUILD_HASH}" 0 12 VEIL_BUILD_ID)
configure_file(Resources/BuildIdentity.h.in "${CMAKE_CURRENT_BINARY_DIR}/BuildIdentity.h" @ONLY)
add_executable(VEIL MACOSX_BUNDLE Sources/main.mm Sources/VEEngine.mm Sources/ProcessResources.mm ${SWIFT_SOURCES})
target_include_directories(VEIL PRIVATE "${CEF_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/Sources" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(VEIL PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:CEF_USE_SANDBOX;CEF_API_VERSION=15600>")
target_compile_options(VEIL PRIVATE
  "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc;-std=c++20>"
  "$<$<COMPILE_LANGUAGE:Swift>:-parse-as-library;-target;arm64-apple-macosx14.0;-import-objc-header;${CMAKE_CURRENT_SOURCE_DIR}/Sources/VEEngine.h>"
)
target_link_options(VEIL PRIVATE -target arm64-apple-macosx14.0)
target_link_libraries(VEIL PRIVATE libcef_dll_wrapper "-framework Cocoa" "-framework QuartzCore" "-framework WebKit" "-framework MediaPlayer" "-framework QuickLookThumbnailing" sqlite3 c++)
set_target_properties(VEIL PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Info.plist" LINKER_LANGUAGE Swift)
set_target_properties(VEIL PROPERTIES XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
set(CEF_HELPER_TARGET VEIL_Helper)
set(CEF_HELPER_OUTPUT_NAME "VEIL Helper")
foreach(suffix ${CEF_HELPER_APP_SUFFIXES})
  string(REPLACE ":" ";" parts ${suffix})
  list(GET parts 0 name_suffix)
  list(GET parts 1 target_suffix)
  list(GET parts 2 plist_suffix)
  set(helper "VEIL_Helper${target_suffix}")
  set(helper_name "VEIL Helper${name_suffix}")
  configure_file(Resources/Helper.plist.in "${CMAKE_CURRENT_BINARY_DIR}/${helper}.plist" @ONLY)
  add_executable(${helper} MACOSX_BUNDLE Sources/helper.cc Sources/Renderer.cc)
  target_include_directories(${helper} PRIVATE "${CEF_ROOT}" Sources)
  target_compile_definitions(${helper} PRIVATE CEF_USE_SANDBOX CEF_API_VERSION=15600)
  target_link_libraries(${helper} PRIVATE libcef_dll_wrapper "-framework Cocoa")
  set_target_properties(${helper} PROPERTIES OUTPUT_NAME "${helper_name}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/${helper}.plist")
  set_target_properties(${helper} PROPERTIES XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
  add_dependencies(VEIL ${helper})
  add_custom_command(TARGET VEIL POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "$<TARGET_BUNDLE_DIR:${helper}>" "$<TARGET_BUNDLE_DIR:VEIL>/Contents/Frameworks/${helper_name}.app" VERBATIM)
endforeach()
add_custom_command(TARGET VEIL POST_BUILD COMMAND ${CMAKE_COMMAND} -E remove_directory "$<TARGET_BUNDLE_DIR:VEIL>/Contents/Frameworks/Chromium Embedded Framework.framework" VERBATIM)
COPY_MAC_FRAMEWORK(VEIL "${CEF_ROOT}/Release" "$<TARGET_BUNDLE_DIR:VEIL>")
add_custom_command(TARGET VEIL POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:VEIL>/Contents/Frameworks" "$<TARGET_BUNDLE_DIR:VEIL>/Contents/Resources"
  COMMAND /usr/bin/xcrun actool "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Icons.xcassets" --compile "$<TARGET_BUNDLE_DIR:VEIL>/Contents/Resources" --platform macosx --minimum-deployment-target 14.0 --output-partial-info-plist "${CMAKE_CURRENT_BINARY_DIR}/asset-info.plist" --app-icon AppIcon --target-device mac --compress-pngs
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Phosphor-LICENSE.txt" "$<TARGET_BUNDLE_DIR:VEIL>/Contents/Resources/Phosphor-LICENSE.txt"
  COMMAND ${CMAKE_COMMAND} -E copy "${CEF_ROOT}/LICENSE.txt" "$<TARGET_BUNDLE_DIR:VEIL>/Contents/Resources/CEF-LICENSE.txt"
  COMMAND ${CMAKE_COMMAND} -E copy "${CEF_ROOT}/CREDITS.html" "$<TARGET_BUNDLE_DIR:VEIL>/Contents/Resources/Chromium-CREDITS.html"
  COMMAND /bin/bash "${CMAKE_CURRENT_SOURCE_DIR}/scripts/sign.sh" "$<TARGET_BUNDLE_DIR:VEIL>"
  VERBATIM)
